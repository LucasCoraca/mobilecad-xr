cmake_minimum_required(VERSION 3.22)
project(mobilecad_xr CXX)
set(CMAKE_CXX_STANDARD 17)
# fcbridge: MobileCAD's JNI bridge to the embedded CPython + FreeCAD (from the mobilecad submodule)
set(PY $ENV{HOME}/Android/fcdeps/${ANDROID_ABI}/python/prefix)
# without MobileCAD's native build tree: python.org's CPython 3.14.4 Android package (same version and soname; only
# headers + link stub are used, the APK ships MobileCAD's libpython), as in mobilecad/app/app/src/main/cpp/CMakeLists.txt
if(NOT EXISTS ${PY}/include/python3.14/Python.h AND EXISTS $ENV{HOME}/Android/pyprefix/${ANDROID_ABI}/prefix/include/python3.14/Python.h)
  set(PY $ENV{HOME}/Android/pyprefix/${ANDROID_ABI}/prefix)
endif()
add_library(fcbridge SHARED ${MOBILECAD_DIR}/app/app/src/main/cpp/fcbridge.cpp)
target_include_directories(fcbridge PRIVATE ${PY}/include/python3.14)
target_link_libraries(fcbridge ${PY}/lib/libpython3.14.so log dl)
# mcxr: the XR core (shared with the desktop build) + Android glue; OpenXR loader from the Khronos AAR (prefab)
find_package(OpenXR REQUIRED CONFIG)
set(CORE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../core)
add_library(mcxr SHARED xr_android.cpp ${CORE}/scene.cpp ${CORE}/renderer.cpp ${CORE}/interaction.cpp ${CORE}/hands.cpp ${CORE}/gltf_model.cpp ${CORE}/xr_session.cpp)
target_include_directories(mcxr PRIVATE ${CORE})
target_link_libraries(mcxr OpenXR::openxr_loader EGL GLESv3 android log)
